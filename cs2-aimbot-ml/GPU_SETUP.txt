# GPU Training Setup - NVIDIA
# =============================
# Für 10-20x schnelleres Training!

# 1. CUDA-kompatibles PyTorch installieren
# -----------------------------------------
# Gehe zu: https://pytorch.org/get-started/locally/
# Wähle:
#   - PyTorch Build: Stable
#   - OS: Windows
#   - Package: Pip
#   - Language: Python
#   - Compute Platform: CUDA 11.8 (oder 12.1 für neuere GPUs)

# Beispiel Installation:
pip3 install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu118

# Für CUDA 12.1 (neuere GPUs):
# pip3 install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu121

# 2. Nach Installation prüfen:
# -----------------------------
python -c "import torch; print('CUDA:', torch.cuda.is_available()); print('GPU:', torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'N/A')"

# Sollte zeigen:
# CUDA: True
# GPU: NVIDIA GeForce RTX 3080 (oder deine GPU)

# 3. Training wird automatisch GPU nutzen!
# -----------------------------------------
# Das train_model.py Script erkennt GPU automatisch und nutzt sie.

# 4. Optimierungen für schnelleres Training
# ------------------------------------------
# In train_model.py kannst du ändern:

# SCHNELLER (weniger genau):
model_size = "n"    # nano - schnellstes
epochs = 50         # weniger Epochs
batch = 32          # größere Batch Size (wenn genug VRAM)

# GENAUER (langsamer):
model_size = "s"    # small - besser
epochs = 150        # mehr Epochs
batch = 16          # kleiner wenn VRAM nicht reicht

# BESTE PERFORMANCE (mit guter GPU):
model_size = "m"    # medium - bestes Balance
epochs = 100        # standard
batch = 32          # größer wenn du >6GB VRAM hast
imgsz = 640         # standard

# 5. Performance erwarten:
# ------------------------
# CPU:     2-3 Stunden für 100 Epochs
# GPU:     10-15 Minuten für 100 Epochs! 🚀
#
# GPU VRAM Empfehlungen:
# 4GB  → batch=8,  model_size="n"
# 6GB  → batch=16, model_size="n" oder "s"
# 8GB+ → batch=32, model_size="s" oder "m"
# 12GB+→ batch=64, model_size="m" oder "l"
